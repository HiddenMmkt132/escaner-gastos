// Pegá esto en Extensiones → Apps Script de tu Google Sheet // Luego: Implementar → Nueva implementación → Web App // Ejecutar como: Yo | Acceso: Cualquier persona function doGet(e) { try { var p = e.parameter; if (!p.mes) return ContentService.createTextOutput("✓ gasto·scan conectado").setMimeType(ContentService.MimeType.TEXT); var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName(p.mes); if (!sheet) { return ContentService.createTextOutput( JSON.stringify({ok: false, error: "Hoja no encontrada: " + p.mes}) ).setMimeType(ContentService.MimeType.JSON); } // Buscar primera fila vacía en columna J (col 10) desde fila 18 var lastRow = sheet.getLastRow(); var targetRow = lastRow + 1; for (var r = 18; r <= lastRow; r++) { var val = sheet.getRange(r, 10).getValue(); if (val === "" || val === null) { targetRow = r; break; } } // Escribir: J=Concepto, K=GastoEsperado, L=GastoReal, M=MetodoPago, N=Fecha, O=Pagado, R=Categoria sheet.getRange(targetRow, 10).setValue(p.concepto || ""); sheet.getRange(targetRow, 11).setValue(parseFloat(p.gastoEsperado) || 0); sheet.getRange(targetRow, 12).setValue(parseFloat(p.gastoReal) || 0); sheet.getRange(targetRow, 13).setValue(p.metodoPago || ""); sheet.getRange(targetRow, 14).setValue(new Date(p.fecha)); sheet.getRange(targetRow, 15).setValue(p.pagado === "TRUE"); sheet.getRange(targetRow, 18).setValue(p.categoria || ""); return ContentService.createTextOutput( JSON.stringify({ok: true, row: targetRow}) ).setMimeType(ContentService.MimeType.JSON); } catch(err) { return ContentService.createTextOutput( JSON.stringify({ok: false, error: err.toString()}) ).setMimeType(ContentService.MimeType.JSON); } }
